# Mac HUD Panel — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Agregar un panel popover al tray icon del desktop Electron que muestra proyecto activo, streaming de Claude, indicador de 6 fases y botón de avanzar fase.

**Architecture:** Nuevo `BrowserWindow` (panel-window.js) + contextBridge (panel-preload.js) + UI HTML (panel.html). Main.js agrega el ítem "📊 Abrir panel →" al menú del tray y conecta los eventos existentes de pty y proyectos al panel.

**Tech Stack:** Electron 30, Node.js, HTML/CSS/JS inline, ipcMain/ipcRenderer

**Spec:** `docs/superpowers/specs/2026-08-15-mac-panel-design.md`

## Global Constraints

- Brand orange: `#f04e23` — usar exactamente este valor, no `#f97316` ni otro
- Fondo del panel: `rgba(20,20,20,0.85)` + `vibrancy: 'hud'`
- Tamaño fijo: `320×480px`, `resizable: false`
- Flag de primer uso: `~/.config/cc-controller/panel-seen` (archivo vacío — existencia = visto)
- Todas las funciones de negocio ya existen en main.js — NO crear nueva lógica, solo exponer
- `contextIsolation: true`, `nodeIntegration: false` en todos los BrowserWindow
- Archivo de flag: usar `writeFileSync(path, '', { flag: 'wx' })` para creación atómica
- No modificar: `bridge.js`, `pty.js`, `projects.js`, `setup-window.js`, `claude-md.js`
- El panel no aparece en Dock ni en cmd+tab (`skipTaskbar: true`)
- PHASES array: `[{n:1,name:'Ideación'},{n:2,name:'POC Local'},{n:3,name:'Lanzamiento'},{n:4,name:'Backend'},{n:5,name:'App Directa'},{n:6,name:'Validación'}]`

### Task 1: panel-window.js — BrowserWindow factory

**Files:**
- Create: `desktop/src/panel-window.js`

**Interfaces:**
- Produces:
  - `createPanel(): BrowserWindow` — crea la ventana, no la muestra
  - `togglePanel(panel, tray): void` — muestra u oculta
  - `positionPanel(panel, tray): void` — posiciona bajo el ícono del tray
  - `sendToPanel(panel, event, data): void` — wrapper seguro para webContents.send
  - `isFirstOpen(): boolean` — true si el flag no existe
  - `markSeen(): void` — crea el archivo flag

- [ ] **Step 1: Crear desktop/src/panel-window.js**

```js
import { BrowserWindow } from 'electron';
import path from 'path';
import { fileURLToPath } from 'url';
import { existsSync, writeFileSync } from 'fs';
import { homedir } from 'os';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const FLAG_PATH = path.join(homedir(), '.config', 'cc-controller', 'panel-seen');

export function createPanel() {
  const win = new BrowserWindow({
    width: 320,
    height: 480,
    frame: false,
    resizable: false,
    alwaysOnTop: true,
    vibrancy: 'hud',
    backgroundColor: '#00000000',
    show: false,
    skipTaskbar: true,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      preload: path.join(__dirname, 'panel-preload.js'),
    },
  });
  win.loadFile(path.join(__dirname, 'panel.html'));
  win.on('blur', () => win.hide());
  return win;
}

export function togglePanel(panel, tray) {
  if (panel.isVisible()) {
    panel.hide();
  } else {
    positionPanel(panel, tray);
    panel.show();
    panel.focus();
  }
}

export function positionPanel(panel, tray) {
  const tb = tray.getBounds();
  const wb = panel.getBounds();
  const x = Math.round(tb.x + tb.width / 2 - wb.width / 2);
  const y = tb.y + tb.height + 4;
  panel.setPosition(x, y);
}

export function sendToPanel(panel, event, data) {
  if (panel && !panel.isDestroyed() && panel.webContents) {
    panel.webContents.send(event, data);
  }
}

export function isFirstOpen() {
  return !existsSync(FLAG_PATH);
}

export function markSeen() {
  try { writeFileSync(FLAG_PATH, '', { flag: 'wx' }); } catch (_) {}
}
```

### Task 2: panel-preload.js — contextBridge API

**Files:**
- Create: `desktop/src/panel-preload.js`

**Interfaces:**
- Consumes: nada de tasks anteriores (solo Electron APIs)
- Produces: `window.cc` expuesto al panel HTML con:
  - `cc.onState(cb)` — suscripción a panel:state
  - `cc.onChunk(cb)` — suscripción a panel:chunk
  - `cc.onStatus(cb)` — suscripción a panel:status
  - `cc.advance(projectId, phase)` — invoke panel:advance-phase
  - `cc.openTerminal(projectId)` — invoke panel:open-terminal
  - `cc.switchProject(id)` — invoke panel:switch-project
  - `cc.openMenu()` — invoke panel:open-menu
  - `cc.markSeen()` — invoke panel:mark-seen
  - `cc.close()` — invoke panel:close

- [ ] **Step 1: Crear desktop/src/panel-preload.js**

```js
import { contextBridge, ipcRenderer } from 'electron';

contextBridge.exposeInMainWorld('cc', {
  onState:       (cb) => ipcRenderer.on('panel:state',  (_, d) => cb(d)),
  onChunk:       (cb) => ipcRenderer.on('panel:chunk',  (_, d) => cb(d)),
  onStatus:      (cb) => ipcRenderer.on('panel:status', (_, d) => cb(d)),
  advance:       (projectId, phase) => ipcRenderer.invoke('panel:advance-phase', projectId, phase),
  openTerminal:  (projectId)        => ipcRenderer.invoke('panel:open-terminal', projectId),
  switchProject: (id)               => ipcRenderer.invoke('panel:switch-project', id),
  openMenu:      ()                 => ipcRenderer.invoke('panel:open-menu'),
  markSeen:      ()                 => ipcRenderer.invoke('panel:mark-seen'),
  close:         ()                 => ipcRenderer.invoke('panel:close'),
});
```

```html
<script>
const PHASES = [
  { n: 1, name: 'Ideación' },
  { n: 2, name: 'POC Local' },
  { n: 3, name: 'Lanzamiento' },
  { n: 4, name: 'Backend' },
  { n: 5, name: 'App Directa' },
  { n: 6, name: 'Validación' },
];

const MAX_LOG_LINES = 500;
let state = { project: null, projects: [], running: false, firstOpen: false };
let logLines = [];
let autoScroll = true;
let hintTimer = null;
let dropdownOpen = false;

const elProjectName  = document.getElementById('project-name');
const elPhasesDots   = document.getElementById('phases-dots');
const elPhaseName    = document.getElementById('phase-name');
const elPhaseCounter = document.getElementById('phase-counter');
const elLogArea      = document.getElementById('log-area');
const elLogEmpty     = document.getElementById('log-empty');
const elScrollBadge  = document.getElementById('scroll-badge');
const elBtnAdvance   = document.getElementById('btn-advance');
const elBtnTerm      = document.getElementById('btn-term');
const elBtnMenu      = document.getElementById('btn-menu');
const elBtnClose     = document.getElementById('btn-close');
const elHint         = document.getElementById('hint');
const elHintClose    = document.getElementById('hint-close');
const elProjectRow   = document.getElementById('project-row');
const elDropdown     = document.getElementById('project-dropdown');

// ── Render ──────────────────────────────────────
function renderState() {
  const { project, projects, running } = state;

  // Project name
  elProjectName.textContent = project ? project.name : 'Sin proyecto activo';

  // Phase dots
  const phase = project?.phase ?? 0;
  const dots = elPhasesDots.querySelectorAll('.dot');
  dots.forEach((dot, i) => {
    dot.classList.toggle('active', i + 1 === phase);
    dot.classList.toggle('done', i + 1 < phase);
  });

  // Phase label
  const phaseInfo = PHASES[phase - 1];
  elPhaseName.textContent  = phaseInfo ? phaseInfo.name : '—';
  elPhaseCounter.textContent = phase ? `Fase ${phase} de 6` : '—';

  // Advance button
  const nextPhase = PHASES[phase];
  const canAdvance = !running && phase > 0 && phase < 6 && !!project;
  elBtnAdvance.disabled = !canAdvance;
  elBtnAdvance.textContent = nextPhase && canAdvance
    ? `Avanzar: ${nextPhase.name} →`
    : phase >= 6 ? 'Validación completa ✓' : 'Avanzar fase →';

  // Terminal button
  elBtnTerm.disabled = !project;

  // Log empty state
  elLogEmpty.style.display = logLines.length === 0 ? 'block' : 'none';
}

function renderDropdown() {
  elDropdown.innerHTML = '';
  state.projects.forEach(p => {
    const item = document.createElement('div');
    item.className = 'project-dropdown-item' + (p.id === state.project?.id ? ' active' : '');
    item.textContent = '📁 ' + p.name;
    item.addEventListener('click', (e) => {
      e.stopPropagation();
      closeDropdown();
      if (p.id !== state.project?.id) cc.switchProject(p.id);
    });
    elDropdown.appendChild(item);
  });
}

// ── Log ──────────────────────────────────────────
function appendChunk(text, _done) {
  if (!text) return;
  elLogEmpty.style.display = 'none';

  // Append to last line or create new
  const lines = text.split('\n');
  lines.forEach((part, i) => {
    if (i === 0 && logLines.length > 0) {
      logLines[logLines.length - 1].text += part;
    } else {
      logLines.push({ text: part, role: 'assistant' });
    }
  });

  // Ring buffer
  if (logLines.length > MAX_LOG_LINES) logLines.splice(0, logLines.length - MAX_LOG_LINES);

  // Rerender log
  rerenderLog();

  if (autoScroll) scrollToBottom();
  else elScrollBadge.classList.add('visible');
}

function rerenderLog() {
  // Remove all log-line elements and re-add
  const existing = elLogArea.querySelectorAll('.log-line');
  existing.forEach(el => el.remove());

  const fragment = document.createDocumentFragment();
  logLines.forEach(line => {
    const div = document.createElement('div');
    div.className = 'log-line ' + (line.role || 'assistant');
    div.textContent = line.text;
    fragment.appendChild(div);
  });
  elLogArea.insertBefore(fragment, elScrollBadge);
}

function scrollToBottom() {
  elLogArea.scrollTop = elLogArea.scrollHeight;
}

elLogArea.addEventListener('scroll', () => {
  const nearBottom = elLogArea.scrollTop + elLogArea.clientHeight >= elLogArea.scrollHeight - 40;
  autoScroll = nearBottom;
  if (nearBottom) elScrollBadge.classList.remove('visible');
});

elScrollBadge.addEventListener('click', () => {
  autoScroll = true;
  scrollToBottom();
  elScrollBadge.classList.remove('visible');
});

// ── Hint ─────────────────────────────────────────
function showHint() {
  elHint.classList.add('visible');
  hintTimer = setTimeout(dismissHint, 8000);
}

function dismissHint() {
  clearTimeout(hintTimer);
  elHint.classList.remove('visible');
  cc.markSeen();
}

elHintClose.addEventListener('click', dismissHint);

// ── Dropdown ──────────────────────────────────────
function toggleDropdown() {
  dropdownOpen = !dropdownOpen;
  elDropdown.classList.toggle('open', dropdownOpen);
  if (dropdownOpen) renderDropdown();
}

function closeDropdown() {
  dropdownOpen = false;
  elDropdown.classList.remove('open');
}

elProjectRow.addEventListener('click', toggleDropdown);
document.addEventListener('click', (e) => {
  if (!elProjectRow.contains(e.target)) closeDropdown();
});

// ── Actions ──────────────────────────────────────
elBtnAdvance.addEventListener('click', () => {
  const { project } = state;
  if (!project || elBtnAdvance.disabled) return;
  cc.advance(project.id, project.phase + 1);
});

elBtnTerm.addEventListener('click', () => {
  if (state.project) cc.openTerminal(state.project.id);
});

elBtnMenu.addEventListener('click', () => cc.openMenu());
elBtnClose.addEventListener('click', () => cc.close());

// ── IPC ──────────────────────────────────────────
cc.onState(({ project, projects, running, firstOpen }) => {
  const projectChanged = project?.id !== state.project?.id;
  state = { project, projects, running, firstOpen };
  if (projectChanged) {
    logLines = [];
    rerenderLog();
    autoScroll = true;
  }
  renderState();
  if (firstOpen) showHint();
});

cc.onChunk(({ text, done }) => appendChunk(text, done));

cc.onStatus(({ status }) => {
  state.running = status === 'running';
  renderState();
});

// Initial render
renderState();
</script>
</body>
</html>
```
